+export function storyMatchesClub(story, club) {
+  if (!club) return false;
+  const keywords = clubKeywords[club.toLowerCase()];

+    story.summary,
+    story.ai_summary,
+    story.source
+  ].filter(Boolean).join(" ").toLowerCase();
+  
+  return keywords.some(keyword => textToSearch.includes(keyword));

+export function filterOldStories(stories, relativeTo = Date.now()) {
+  const maxAgeMs = 24 * 60 * 60 * 1000; // 24 hours
+  return stories.filter(story => {
+    if (!story.publishedAt) return true;
+    const pubDate = new Date(story.publishedAt);
+    if (isNaN(pubDate.getTime())) return true;
+    return (relativeTo - pubDate.getTime()) <= maxAgeMs;
+  });
